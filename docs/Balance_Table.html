Balance Table

# Load the following packages -------------
packages <- c("modelsummary", 
              "dplyr", 
              "haven")

if (!requireNamespace("pacman", quietly = TRUE)) install.packages("pacman") 
pacman::p_load(packages, character.only = TRUE) # switch to TRUE to install packages 

# Load the sample data set --------------
data <- read_dta("/Users/koniks/Downloads/nlsw88.dta")
dir.create("Outputs", recursive = TRUE, showWarnings = TRUE) # to create a new folder 

# Create the data frame --------------
df_balance <- data %>% 
  select('Age in current year' = age, # replace the following with variable names 
         'Married' = married,
         'Union worker' = union, 
         'Hourly wage' = wage,
         'South' = south,
         collgrad # replace with variable indicating treatment status
         )

# Run the balance test and produce the table --------------
datasummary_balance(
  ~collgrad,
  data = df_balance,
  stars = c('*' = .05, '**' = .01, '***' = .001),
  title = "Balance by college graduation status (as treatment)",
  note = "Comparing college graduates vs non-graduates: Includes HHs with observations for endline and baseline,
* p < 0.05, ** p < 0.01, *** p < 0.001", 
  output = "Outputs/balance.tex"
)
